\section*{23 - thermal storage and resistance}

\[
\begin{array}{ll}x=T-T_{\mathrm b}\ [\mathrm K]&\text{temperature rise}\\[0.6em]C\ [\mathrm{J/K}]&\text{thermal capacitance}\\[0.6em]R\ [\mathrm{K/W}]&\text{thermal resistance}\\[0.6em]P_{\mathrm{in}},P_{\mathrm{out}}\ [\mathrm W]&\text{heat flow rates}\\[0.6em]U_{\mathrm{th}}\ [\mathrm J]&\text{excess thermal energy}\end{array}
\]

\subsection*{stored energy}
\[
\displaystyle U_{\mathrm{th}}=Cx
\]

\subsection*{heat balance}
\[
\displaystyle \frac{dU_{\mathrm{th}}}{dt}=P_{\mathrm{in}}-P_{\mathrm{out}},\qquad P_{\mathrm{out}}=\frac xR
\]

\subsection*{first-order thermal response}
\[
\displaystyle C\dot x=P_{\mathrm{in}}-\frac xR,\qquad\tau=RC
\]

\subsection*{steady rise}
\[
\displaystyle x_*=RP_{\mathrm{in}}
\]
