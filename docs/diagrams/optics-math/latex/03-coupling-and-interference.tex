\section*{03 - coupling and interference}

\[
\begin{array}{ll}E_1,E_2\ [\sqrt{\mathrm W}]&\text{coherent fields}\\[0.6em]\psi\ [\mathrm{rad}]&\text{relative phase}\\[0.6em]t,k\ [1]&\text{coupler coefficients}\\[0.6em]b,u\ [\sqrt{\mathrm W}]&\text{bus, ring fields}\\[0.6em]\mathbf C\ [1]&\text{coupler matrix}\end{array}
\]

\subsection*{field addition}
\[
\displaystyle \begin{aligned}E_{\mathrm{out}}&=E_1+E_2e^{i\psi}\\P_{\mathrm{out}}&=|E_1|^2+|E_2|^2\\&\quad+2\operatorname{Re}\!\left(E_1^*E_2e^{i\psi}\right)\end{aligned}
\]

\subsection*{lossless coupler}
\[
\displaystyle \mathbf C=\begin{pmatrix}t&-ik\\-ik&t\end{pmatrix},\qquad\begin{pmatrix}b_{\mathrm{out}}\\u_{\mathrm{out}}\end{pmatrix}=\mathbf C\begin{pmatrix}b_{\mathrm{in}}\\u_{\mathrm{in}}\end{pmatrix}
\]

\subsection*{conservation}
\[
\displaystyle t^2+k^2=1,\qquad \mathbf C^{\dagger}\mathbf C=\mathbf I
\]
