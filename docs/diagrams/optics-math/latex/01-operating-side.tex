\section*{01 - transmission and operating side}

\[
\begin{array}{ll}\lambda_L,\lambda_r\ [\mathrm m]&\text{laser, resonance}\\[0.6em]\gamma\ [\mathrm m]&\text{Lorentzian HWHM}\\[0.6em]\phi\ [\mathrm{rad}]&\text{round-trip phase}\\[0.6em]a\ [1]&\text{round-trip field retention}\\[0.6em]t_j,k_j\ [1]&\text{coupler coefficients}\\[0.6em]h_{\mathrm{th}}\ [1]&\text{through field ratio}\\[0.6em]\mathcal T,\mathcal D\ [1]&\text{through, drop fractions}\end{array}
\]

\subsection*{zero backscatter}
\[
\displaystyle r_b=0,\qquad r=a t_1t_2
\]

\subsection*{Airy response}
\[
\displaystyle \begin{aligned}h_{\mathrm{th}}(\phi)&=\frac{t_1-a t_2e^{-i\phi}}{1-r e^{-i\phi}}\\\mathcal T(\phi)&=|h_{\mathrm{th}}(\phi)|^2\\\mathcal D(\phi)&=\frac{k_1^2k_2^2a}{|1-r e^{-i\phi}|^2}\end{aligned}
\]

\subsection*{stable-side convention}
\[
\displaystyle \delta\lambda=\lambda_r-\lambda_L>0
\]

\subsection*{Lorentzian maximum slope}
\[
\displaystyle \begin{aligned}x&=\frac{\lambda_L-\lambda_r}{\gamma},\qquad\frac{\mathcal D}{\mathcal D_{\mathrm{pk}}}=\frac1{1+x^2}\\|x|_{\mathrm{max\ slope}}&=\frac1{\sqrt3},\qquad\frac{\mathcal D}{\mathcal D_{\mathrm{pk}}}=\frac34\end{aligned}
\]
