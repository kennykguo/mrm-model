\section*{00 - ports and travelling fields}

\[
\begin{array}{ll}s_j,\ o_j\ [\sqrt{\mathrm W}]&\text{incoming, outgoing}\\[0.6em]u,v\ [\sqrt{\mathrm W}]&\text{CW, CCW fields}\\[0.6em]j\in\{0,1,2,3\}&\text{port index}\\[0.6em]k_j,t_j\ [1]&\text{cross, self coupling}\\[0.6em]\mathbf S\ [1]&\text{scattering matrix}\end{array}
\]

\subsection*{ports}
\[
\displaystyle \begin{aligned}0&\quad\text{input}\\1&\quad\text{through}\\2&\quad\text{drop}\\3&\quad\text{add}\end{aligned}
\]

\subsection*{field vectors}
\[
\displaystyle \mathbf x=\begin{pmatrix}u\\v\end{pmatrix},\qquad \mathbf o=\mathbf S\mathbf s
\]

\subsection*{power}
\[
\displaystyle P_j=|s_j|^2,\qquad P_{\mathrm{out},j}=|o_j|^2
\]

\subsection*{couplers}
\[
\displaystyle t_j^2+k_j^2=1
\]
