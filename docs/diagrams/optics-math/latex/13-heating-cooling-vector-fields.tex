\section*{13 - thermal vector field and stability}

\[
\begin{array}{ll}T,T_{\mathrm b}\ [\mathrm K]&\text{ring, base temperature}\\[0.6em]C\ [\mathrm{J/K}]&\text{thermal capacitance}\\[0.6em]R\ [\mathrm{K/W}]&\text{thermal resistance}\\[0.6em]P_h,P_{\mathrm{abs}}\ [\mathrm W]&\text{heater, absorbed power}\\[0.6em]f(T)\ [\mathrm{K/s}]&\text{temperature vector field}\\[0.6em]T_*\ [\mathrm K]&\text{equilibrium}\end{array}
\]

\subsection*{single thermal state}
\[
\displaystyle C\dot T=P_h+P_{\mathrm{abs}}(T)-\frac{T-T_{\mathrm b}}R
\]

\subsection*{heating gap}
\[
\displaystyle \begin{aligned}P_{\mathrm{req}}(T)&=\frac{T-T_{\mathrm b}}R-P_h\\\dot T=f(T)&=\frac{P_{\mathrm{abs}}(T)-P_{\mathrm{req}}(T)}C\end{aligned}
\]

\subsection*{equilibrium and local stability}
\[
\displaystyle \begin{aligned}f(T_*)&=0\\f'(T_*)&<0\quad\text{stable}\\f'(T_*)&>0\quad\text{unstable}\end{aligned}
\]

\subsection*{direction}
\[
\displaystyle f(T)>0\Rightarrow\dot T>0,\qquad f(T)<0\Rightarrow\dot T<0
\]
