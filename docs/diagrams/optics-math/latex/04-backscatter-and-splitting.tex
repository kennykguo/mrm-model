\section*{04 - backscatter and resonance splitting}

\[
\begin{array}{ll}r_{b,n}\ [1]&\text{round-trip backscatter}\\[0.6em]\rho_n,\rho\ [1]&\text{scatterer reflection}\\[0.6em]\tau\ [1]&\text{scatterer transmission}\\[0.6em]\theta\ [\mathrm{rad}]&\text{phase within an order}\\[0.6em]\phi\ [\mathrm{rad}]&\text{round-trip phase}\\[0.6em]g\ [\mathrm{rad/s}]&\text{mode coupling rate}\\[0.6em]\omega_0,\omega_\pm\ [\mathrm{rad/s}]&\text{resonant frequencies}\\[0.6em]T_{\mathrm{rt}}\ [\mathrm s]&\text{round-trip group delay}\end{array}
\]

\subsection*{equivalent scatterers}
\[
\displaystyle \begin{aligned}\phi&=2\pi n+\theta,\quad n\in\mathbb Z,\quad0\le\theta<2\pi\\\rho_n&=-\frac{i}{2}(-1)^n r_{b,n}^*,\qquad\tau=\sqrt{1-|\rho|^2}\end{aligned}
\]

\subsection*{between orders}
\[
\displaystyle \begin{aligned}\rho(\phi)&=\cos^2\!\left(\frac\theta4\right)\rho_n+\sin^2\!\left(\frac\theta4\right)\rho_{n+1}\\\frac{d\rho}{d\phi}&=\frac14\sin\!\left(\frac\theta2\right)(\rho_{n+1}-\rho_n)\end{aligned}
\]

\subsection*{near-resonance TCMT}
\[
\displaystyle g=\frac{r_{b,n}}{T_{\mathrm{rt}}},\qquad\omega_\pm=\omega_0\pm|g|
\]
