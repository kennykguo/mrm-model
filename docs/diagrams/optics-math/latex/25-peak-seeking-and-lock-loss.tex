\section*{25 - self-stabilization and fold sensitivity}

\[
\begin{array}{ll}P_{\mathrm{abs}}\ [\mathrm W]&\text{absorbed optical power}\\[0.6em]T,T_{\mathrm b}\ [\mathrm K]&\text{ring, base temperature}\\[0.6em]P_h\ [\mathrm W]&\text{heater power}\\[0.6em]R\ [\mathrm{K/W}]&\text{thermal resistance}\\[0.6em]\lambda_r,\lambda_L\ [\mathrm m]&\text{resonance, laser wavelength}\\[0.6em]\mathcal S\ [1]&\text{thermal suppression factor}\end{array}
\]

\subsection*{single-state steady sensitivity}
\[
\displaystyle \frac{dT_*}{dP_h}=\frac R{1-RP'_{\mathrm{abs}}(T_*)}
\]

\subsection*{negative optical feedback}
\[
\displaystyle \begin{aligned}\lambda_r&>\lambda_L,\qquad P'_{\mathrm{abs}}(T_*)<0\\\mathcal S&=1-RP'_{\mathrm{abs}}(T_*)>1\end{aligned}
\]

\subsection*{nominal suppression}
\[
\displaystyle \mathcal S\sim3\text{--}7,\qquad\frac{dT_*}{dP_h}=\frac R{\mathcal S}
\]

\subsection*{fold}
\[
\displaystyle 1-RP'_{\mathrm{abs}}(T_*)=0,\qquad\left|\frac{dT_*}{dP_h}\right|\to\infty
\]
