\section*{02 - resonance orders}

\[
\begin{array}{ll}L\ [\mathrm m]&\text{ring length}\\[0.6em]n_{\mathrm{eff}},n_g\ [1]&\text{effective, group index}\\[0.6em]m,n\in\mathbb Z&\text{longitudinal orders}\\[0.6em]c\ [\mathrm{m/s}]&\text{vacuum light speed}\\[0.6em]T_{\mathrm{rt}}\ [\mathrm s]&\text{round-trip group delay}\end{array}
\]

\subsection*{round-trip resonance}
\[
\displaystyle \phi(\lambda_m)=\frac{2\pi n_{\mathrm{eff}}(\lambda_m)L}{\lambda_m}=2\pi m
\]

\subsection*{group delay and spacing}
\[
\displaystyle \begin{aligned}n_g&=n_{\mathrm{eff}}-\lambda\frac{dn_{\mathrm{eff}}}{d\lambda}\\T_{\mathrm{rt}}&=\frac{n_gL}{c}\\\mathrm{FSR}_{\nu}&\simeq\frac1{T_{\mathrm{rt}}},\qquad\mathrm{FSR}_{\lambda}\simeq\frac{\lambda^2}{n_gL}\end{aligned}
\]

\subsection*{order and residual phase}
\[
\displaystyle \begin{aligned}n&=\left\lfloor\frac{\phi}{2\pi}\right\rfloor\\\theta&=\phi-2\pi n\in[0,2\pi)\end{aligned}
\]
