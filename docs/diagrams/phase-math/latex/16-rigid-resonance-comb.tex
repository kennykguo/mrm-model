\section*{rigid resonance comb}
\[
\begin{array}{ll}m,m_*\in\mathbb Z&\text{resonance order, anchored order}\\[0.6em]\lambda_m^0,\lambda_m^{\rm rigid},\lambda_m^{\rm full}\ [\mathrm m]&\text{reference, translated, fuller resonance}\\[0.6em]\Delta\lambda_r,\epsilon_m\ [\mathrm m]&\text{common shift, comparison error}\\[0.6em]L(T)\ [\mathrm m],\quad n_{\rm eff}\ [1]&\text{temperature-dependent length, index}\end{array}
\]
\subsection*{runtime translation}
\[
\displaystyle \lambda_m^{\rm rigid}=\lambda_m^0+\Delta\lambda_r
\]
\subsection*{wavelength spacing}
\[
\displaystyle \lambda_{m+1}^{\rm rigid}-\lambda_m^{\rm rigid}=\lambda_{m+1}^0-\lambda_m^0
\]
\subsection*{fuller resonance condition}
\[
\displaystyle n_{\rm eff}(\lambda_m^{\rm full},T,V)L(T)=m\lambda_m^{\rm full}
\]
\subsection*{anchored comparison}
\[
\displaystyle \begin{aligned}\epsilon_m&=\lambda_m^{\rm full}-\lambda_m^{\rm rigid}\\\lambda_{m_*}^{\rm full}=\lambda_{m_*}^{\rm rigid}&\quad\Longrightarrow\quad\epsilon_{m_*}=0\\\epsilon_{m_*}=0&\quad\not\Longrightarrow\quad\epsilon_m=0\quad(m\ne m_*)\end{aligned}
\]
