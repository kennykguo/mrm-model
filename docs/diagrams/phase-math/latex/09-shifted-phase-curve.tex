\section*{shifted reference phase}
\[
\begin{array}{ll}\lambda_L,\lambda_0\ [\mathrm m]&\text{laser, reference resonance}\\[0.6em]\Delta\lambda_r\ [\mathrm m]&\text{total resonance shift}\\[0.6em]x,d\ [\mathrm m]&\text{reference coordinate, offset}\\[0.6em]n_{\rm cold}(x)\ [1]&\text{reference effective index}\\[0.6em]L\ [\mathrm m],\quad m\in\mathbb Z&\text{ring length, resonance order}\\[0.6em]\phi,\psi_{\rm add}\ [\mathrm{rad}]&\text{round-trip phase, added coupler phase}\end{array}
\]
\subsection*{coordinates}
\[
\displaystyle x=\lambda_L-\Delta\lambda_r,\qquad d=x-\lambda_0
\]
\subsection*{translated curve}
\[
\displaystyle \phi(\lambda_L;\Delta\lambda_r)=\phi_{\rm ref}(x)=\frac{2\pi L n_{\rm cold}(x)}{x}
\]
\subsection*{resonance}
\[
\displaystyle \lambda_L=\lambda_0+\Delta\lambda_r\quad\Longrightarrow\quad x=\lambda_0,\quad\phi=2\pi m
\]
\subsection*{coupler phase}
\[
\displaystyle \psi_{\rm add}=0
\]
