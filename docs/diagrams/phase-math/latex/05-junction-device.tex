\section*{junction coverage}
\[
\begin{array}{ll}V\ [\mathrm V],\quad f\in[0,1]&\text{reverse bias, junction coverage}\\[0.6em]L,L_J\ [\mathrm m]&\text{ring, junction length}\\[0.6em]q(V),n_J(V)\ [1]&\text{table, ring index contribution}\\[0.6em]\alpha_{\rm FCA},\alpha_J\ [\mathrm{m^{-1}}]&\text{table, ring junction loss}\\[0.6em]c'(V)\ [\mathrm{F/m}]&\text{table capacitance density}\\[0.6em]r'(V)\ [\Omega\,\mathrm m]&\text{table resistance-length product}\\[0.6em]C_J\ [\mathrm F],\quad R_s\ [\Omega]&\text{junction capacitance, resistance}\end{array}
\]
\subsection*{coverage}
\[
\displaystyle L_J=fL
\]
\subsection*{optical properties}
\[
\displaystyle n_J(V)=fq(V),\qquad\alpha_J(V)=f\alpha_{\rm FCA}(V)
\]
\subsection*{electrical properties}
\[
\displaystyle C_J(V)=L_Jc'(V),\qquad R_s(V)=\begin{cases}r'(V)/L_J&L_J>0\\+\infty&L_J=0\end{cases}
\]
