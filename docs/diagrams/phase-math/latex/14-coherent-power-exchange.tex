\section*{coherent power exchange}
\[
\begin{array}{ll}z,\ell\ [\mathrm m]&\text{propagation distance, interaction length}\\[0.6em]A(z),B(z)\ [1]&\text{normalized guide amplitudes}\\[0.6em]\kappa_c\ [\mathrm{m^{-1}}]&\text{spatial amplitude coupling}\\[0.6em]\Delta\beta\ [\mathrm{m^{-1}}]&\text{propagation mismatch}\\[0.6em]C\ [1],\quad i^2=-1&\text{transferred power fraction}\end{array}
\]
\subsection*{phase-matched, lossless}
\[
\displaystyle \Delta\beta=0,\qquad\frac{dA}{dz}=-i\kappa_c B,\qquad\frac{dB}{dz}=-i\kappa_c A
\]
\subsection*{unit launch}
\[
\displaystyle A(0)=1,\quad B(0)=0,\qquad A''+\kappa_c^2A=0
\]
\subsection*{fields}
\[
\displaystyle A(z)=\cos(\kappa_c z),\qquad B(z)=-i\sin(\kappa_c z)
\]
\subsection*{power}
\[
\displaystyle C=|B(\ell)|^2=\sin^2(\kappa_c\ell),\qquad|A|^2+|B|^2=1
\]
